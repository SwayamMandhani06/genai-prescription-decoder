def postprocess_extracted_medicine(med: ExtractedMedicine) -> ExtractedMedicine:
    """
    Applies conservative field-level normalization across an individual medicine line item:
    - Normalizes Indian posology frequencies (1-0-1 -> BD, 1-1-1 -> TDS, etc.) while preserving raw_frequency
    - Normalizes Indian posology duration fractions (5/7 -> 5 days, 2/52 -> 2 weeks, etc.) while preserving raw_duration
    - Corroborates numeric brand suffixes (Augmentin 625, Dolo 650) against CDSCO formulary without fabricating units
    - Preserves all visual metadata (line_index, raw_transcription, evidence)
    - Enforces Zero Fabrication Rule for route: never defaults to 'Oral' unless written or implied by Tab./Cap.
    """
    norm_name = postprocess_extracted_field(med.medicine_name)
    norm_dosage = postprocess_extracted_field(med.dosage)
    norm_freq = postprocess_extracted_field(med.frequency)
    norm_dur = postprocess_extracted_field(med.duration)

    # 1. Numeric Brand Suffix Corroboration against CDSCO Approved Drugs
    raw_strength = normalize_whitespace(med.strength)
    if norm_name.value:
        clean_brand, candidate_strength = corroborate_numeric_brand_suffix(norm_name.value)
        norm_name.value = clean_brand
        if not raw_strength and candidate_strength:
            raw_strength = candidate_strength

    # Strip trailing formulation words from dosage (e.g. '500 mg tablet' -> '500 mg')
    if norm_dosage.value:
        clean_dose = re.sub(
            r'\s+(tablet|tablets|tab|tabs|capsule|capsules|cap|caps)\b',
            '',
            norm_dosage.value,
            flags=re.IGNORECASE,
        ).strip()
        norm_dosage.value = clean_dose if clean_dose else None

    # Bare-Number Dosage / Unitless Dosage Resolution:
    # If norm_dosage.value after stripping formulation words is missing/empty, just a unitless digit (e.g. "1", "2"),
    # or lacks a recognized strength unit (mg, mcg, g, ml, iu, %), AND raw_strength contains a valid strength unit,
    # set norm_dosage.value = raw_strength.
    if raw_strength:
        has_strength_unit = bool(re.search(r'\b(mg|mcg|g|ml|iu|%)\b', raw_strength, re.IGNORECASE))
        if has_strength_unit:
            dosage_is_empty = not norm_dosage.value or norm_dosage.value.upper() in ("NONE", "N/A", "MISSING")
            dosage_is_bare_digit = bool(re.match(r'^\d+(?:\.\d+)?$', str(norm_dosage.value).strip())) if norm_dosage.value else False
            dosage_lacks_unit = not bool(re.search(r'\b(mg|mcg|g|ml|iu|%)\b', str(norm_dosage.value), re.IGNORECASE)) if norm_dosage.value else False

            if dosage_is_empty or dosage_is_bare_digit or dosage_lacks_unit:
                norm_dosage.value = raw_strength
                norm_dosage.status = "confident"
                norm_dosage.presence = "present"
                norm_dosage.extraction_state = "extracted"
        elif not norm_dosage.value or norm_dosage.value.upper() in ("NONE", "N/A", "MISSING"):
            norm_dosage.value = raw_strength
            norm_dosage.status = "confident"
            norm_dosage.presence = "present"
            norm_dosage.extraction_state = "extracted"
    elif (not norm_dosage.value or norm_dosage.value.upper() in ("NONE", "N/A", "MISSING")) and raw_strength:
        norm_dosage.value = raw_strength
        norm_dosage.status = "confident"
        norm_dosage.presence = "present"
        norm_dosage.extraction_state = "extracted"

    # 2. Indian Frequency Normalization (preserving raw_frequency)
    raw_freq_token = med.raw_frequency or norm_freq.value
    raw_freq_token, norm_freq_plain = normalize_indian_frequency(raw_freq_token)
    if norm_freq.value and norm_freq_plain and norm_freq.value != norm_freq_plain:
        norm_freq.value = norm_freq_plain

    # 3. Indian Duration Normalization (preserving raw_duration)
    raw_dur_token = med.raw_duration or norm_dur.value
    raw_dur_token, norm_dur_plain = normalize_indian_duration(raw_dur_token)
    if norm_dur.value and norm_dur_plain and norm_dur.value != norm_dur_plain:
        norm_dur.value = norm_dur_plain

    norm_abbrevs = []
    for a in med.abbreviations:
        c_a = normalize_whitespace(a)
        if c_a and c_a not in norm_abbrevs:
            norm_abbrevs.append(c_a)

    # Add canonical posology abbreviations if recognized
    if raw_freq_token:
        for f_part in raw_freq_token.replace("-", " ").split():
            if f_part.upper() in ("BD", "TDS", "OD", "HS", "SOS", "PRN", "BBF", "PC", "AC", "QID"):

                if up_tok not in norm_abbrevs:
                    norm_abbrevs.append(up_tok)

    route_raw = med.route.value if isinstance(med.route, ExtractedField) else med.route
    route_val = normalize_whitespace(route_raw)

    dosage_form_raw = med.dosage_form.value if isinstance(med.dosage_form, ExtractedField) else (med.dosage_form or med.formulation)

    formulation = normalize_whitespace(med.formulation or dosage_form_raw)

    if route_val is None:
        raw_tx = med.raw_transcription or ""
        spec_inst = med.special_instructions or ""
        combined_text = f"{dosage_form or ''} {formulation or ''} {' '.join(norm_abbrevs)} {norm_name.value or ''} {raw_tx} {spec_inst}".lower()

            route_val = "Injectable"
        elif any(w in combined_text for w in ("eye drop", "eye drops", "ophthalmic")):

        elif any(w in combined_text for w in ("ointment", "cream", "gel", "lotion", "topical")):

        combined_text = f"{formulation or ''} {' '.join(norm_abbrevs)} {norm_name.value or ''}".lower()
        if any(w in combined_text for w in ("cap", "capsule")):
            dosage_form = "Capsule"
        elif any(w in combined_text for w in ("syrup", "syp")):
            dosage_form = "Syrup"
        elif any(w in combined_text for w in ("tab", "tablet")):
            dosage_form = "Tablet"

    # Determine status conservatively
    item_status = med.status

    return ExtractedMedicine(
        medicine_name=norm_name,
        dosage=norm_dosage,
        frequency=norm_freq,
        duration=norm_dur,

    Detects critical unit confusion (e.g., mg vs mcg) and 10x decimal errors (e.g., 5000 mg vs 500 mg).
    Never mutates, alters, or infers posology.
    """
    if not observed_dosage or not observed_dosage.strip():
        return "not_evaluated", None
    if not reference_strength or not reference_strength.strip():
        return "not_evaluated", None

    obs_val, obs_unit = parse_strength_value_unit(observed_dosage)
    ref_val, ref_unit = parse_strength_value_unit(reference_strength)

    # 1. Critical Clinical Safety Bounds (FMEA-01 / HIGH-04)
    if obs_val is not None and ref_val is not None and obs_unit and ref_unit:
        # Unit confusion hazard: mg prescribed instead of microgram (1000x overdose)
        if obs_unit in ("mg", "g") and ref_unit in ("mcg", "ug", "microgram", "micrograms"):

            )

class PlattScalingCalibrator(BaseCalibrator):
    """
    Parametric Logistic Calibration (Platt Scaling):
    P(y=1 | s) = 1 / (1 + exp(-(A * s + B)))
    Fits scalar slope A and intercept B by minimizing binary cross-entropy.
    """

    def __init__(
        self,
        model_id: str = "cal_platt_v1",
        version: str = "1.0.0",
        min_samples: int = 15,
    ):
        super().__init__(
            model_id=model_id,
            method_name="platt_scaling",
            version=version,
            min_samples=min_samples,
        )
        self.slope_a: float = 1.0
        self.intercept_b: float = 0.0

    def fit(self, scores: List[float], labels: List[int]) -> "PlattScalingCalibrator":
        n = len(scores)
        if n < self.min_samples:
            self.is_fitted = False
            self.sample_count = n
            raise InsufficientDataError(
                f"Sample count ({n}) is below minimum threshold ({self.min_samples}) for reliable Platt scaling."
            )

        X = np.array(scores, dtype=np.float64)
        y = np.array(labels, dtype=np.float64)

        # Loss function: Binary Cross Entropy with L2 regularization
        def loss_func(params):
            a, b = params
            logits = a * X + b
            # numerically stable sigmoid log-loss
            # loss = log(1 + exp(logits)) - y * logits
            # Use np.logaddexp(0, logits) for stability
            bce = np.mean(np.logaddexp(0, logits) - y * logits)
            reg = 1e-4 * (a ** 2 + b ** 2)
            return bce + reg

        initial_params = [1.0, 0.0]
        res = minimize(loss_func, initial_params, method="L-BFGS-B")

        self.slope_a = float(res.x[0])
        self.intercept_b = float(res.x[1])
        self.parameters = {
            "slope_a": round(self.slope_a, 6),
            "intercept_b": round(self.intercept_b, 6),
            "loss": round(float(res.fun), 6),
        }
        self.is_fitted = True
        self.sample_count = n
        return self

    def predict_one(self, score: float) -> float:
        if not self.is_fitted:
            return float(min(max(score, 0.0), 1.0))
        logit = self.slope_a * score + self.intercept_b
        # Numerically safe sigmoid
        if logit >= 0:
            prob = 1.0 / (1.0 + math.exp(-logit))
        else:
            prob = math.exp(logit) / (1.0 + math.exp(logit))
        return min(max(prob, 0.0), 1.0)

class NumericFidelityValidator:
    """
    Validates that numbers, decimals, and metric quantities from source prescription
    data are preserved verbatim without mutation, omission, or hallucinated numbers.
    """

    @staticmethod
    def extract_numbers_and_units(text: str) -> List[str]:
        """
        Extracts numbers, decimals, and attached units (e.g. '500 mg', '2.5 mL', '5 days').
        """
        if not text:
            return []
        # Find patterns like 500, 2.5, 650 mg, 5 days, 1-0-1, etc.
        pattern = r"\b\d+(?:\.\d+)?\s*(?:mg|g|ml|mcg|iu|tablets?|capsules?|days?|weeks?|months?|दिन|दिवस)?\b"
        matches = re.findall(pattern, text, flags=re.IGNORECASE)
        return [m.strip().lower() for m in matches if m.strip()]

    @classmethod
    def validate(
        cls,
        source_facts: Dict[str, Optional[str]],
        generated_text: str,
        status: ExplanationEligibilityStatus,
    ) -> Tuple[bool, List[str]]:
        """
        Verifies numeric fidelity between source facts and generated text.
        """
        if status in ("abstained", "unavailable"):
            # In abstained/unavailable states, medication posology is suppressed
            return True, []

        details: List[str] = []
        is_valid = True

        # Extract numeric tokens from source
        dosage_str = source_facts.get("dosage") or ""
        duration_str = source_facts.get("duration") or ""

        # Extract raw numbers from dosage (e.g. 500, 2.5)
        source_dosage_nums = re.findall(r"\b\d+(?:\.\d+)?\b", dosage_str)
        source_duration_nums = re.findall(r"\b\d+(?:\.\d+)?\b", duration_str)

        # Check dosage numbers exist in generated text if dosage was provided
        for num in source_dosage_nums:
            # Word boundary regex ensuring exact number match
            num_pattern = re.compile(rf"(?<![\d.]){re.escape(num)}(?![\d.])")
            if not num_pattern.search(generated_text):
                is_valid = False
                details.append(f"Numeric mutation or omission: Dosage number '{num}' missing from explanation text.")

        # Check duration numbers exist in generated text if duration was provided
        for num in source_duration_nums:
            num_pattern = re.compile(rf"(?<![\d.]){re.escape(num)}(?![\d.])")
            if not num_pattern.search(generated_text):
                is_valid = False
                details.append(f"Numeric mutation or omission: Duration number '{num}' missing from explanation text.")

        return is_valid, details

class SlidingWindowRateLimiter:
    """
    Thread-safe in-memory sliding window rate limiter.
    Tracks timestamps of incoming requests per client key.
    Automatically purges expired timestamps to maintain bounded memory footprint.
    """

    def __init__(self, max_requests: int = 60, window_seconds: int = 60):
        self.max_requests = max_requests
        self.window_seconds = window_seconds
        self._history: Dict[str, List[float]] = {}
        self._last_cleanup = time.time()

    def _cleanup_old_entries(self, current_time: float) -> None:
        """Evicts client keys that have no timestamps in the current window."""
        expired_keys = []
        for key, timestamps in self._history.items():
            valid_timestamps = [t for t in timestamps if current_time - t < self.window_seconds]
            if not valid_timestamps:
                expired_keys.append(key)
            else:
                self._history[key] = valid_timestamps
        for key in expired_keys:
            self._history.pop(key, None)

    def is_allowed(self, client_id: str) -> Tuple[bool, int, int]:
        """
        Evaluates whether a request from client_id is permitted under current quota.
        Returns:
            (is_allowed: bool, remaining_quota: int, retry_after_seconds: int)
        """
        now = time.time()

        # Evict stale entries periodically every 60 seconds or if map exceeds 5000 keys
        if (now - self._last_cleanup > 60) or (len(self._history) > 5000):
            self._cleanup_old_entries(now)
            self._last_cleanup = now

        timestamps = self._history.get(client_id, [])
        valid_timestamps = [t for t in timestamps if now - t < self.window_seconds]

        if len(valid_timestamps) >= self.max_requests:
            oldest_in_window = valid_timestamps[0]
            retry_after = max(1, int(self.window_seconds - (now - oldest_in_window)))
            self._history[client_id] = valid_timestamps
            return False, 0, retry_after

        valid_timestamps.append(now)
        self._history[client_id] = valid_timestamps
        remaining = max(0, self.max_requests - len(valid_timestamps))
        return True, remaining, 0

    def reset(self) -> None:
        """Clears client history for testing and state resets."""
        self._history.clear()
